cmake_minimum_required(VERSION 3.20)
project(v821_sdk_event_regression C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT AK)
    message(FATAL_ERROR "Pass -DAK=<isolated agentic-kit source with dependencies>")
endif()
set(AGENTIC_KIT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AGENTIC_KIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${AK}" sdk EXCLUDE_FROM_ALL)
enable_testing()
add_executable(test_event_inheritance test_event_inheritance.c)
target_include_directories(test_event_inheritance PRIVATE "${AK}/modules/rtc-tcp-client/src")
target_link_libraries(test_event_inheritance PRIVATE tuya_ai_tcp agentic_kit_pal)
target_compile_options(test_event_inheritance PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME sdk_late_control_event COMMAND test_event_inheritance)
